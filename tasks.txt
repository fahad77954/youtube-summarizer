# Getting the id of the video

# streamlit, youtube-transcript-api, groq

def get_id(url):
    match = re.search(
        r"^https?://(www\.|m\.)?(youtube.com/|youtu.be/)(watch\?.*v|shorts|embed|live)?(=|/)?(?P<id>[a-zA-Z0-9_-]{11}).*",
        url,
    )

    if match:
        return match.group("id")
    else:
        return "Failed to find ID"
